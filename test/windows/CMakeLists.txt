cmake_minimum_required(VERSION 3.14)
project(lanflow_clipboard_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(clipboard_test clipboard_test.cpp ../../platforms/windows/runner/clipboard_bridge.cpp)
target_compile_definitions(clipboard_test PRIVATE NOMINMAX)
target_link_libraries(clipboard_test PRIVATE windowscodecs ole32 gdi32 user32)
enable_testing()
add_test(NAME native_clipboard_image_conversion COMMAND clipboard_test)
set_tests_properties(native_clipboard_image_conversion PROPERTIES TIMEOUT 10)
add_executable(window_geometry_test window_geometry_test.cpp)
target_link_libraries(window_geometry_test PRIVATE user32)
add_test(NAME restored_window_fits_monitor COMMAND window_geometry_test)
add_executable(file_picker_test file_picker_test.cpp ../../platforms/windows/runner/file_picker_bridge.cpp)
target_compile_definitions(file_picker_test PRIVATE NOMINMAX UNICODE _UNICODE)
target_compile_options(file_picker_test PRIVATE /W4 /WX /wd4100 /EHsc /utf-8)
target_link_libraries(file_picker_test PRIVATE ole32 shell32 user32)
add_test(NAME asynchronous_file_picker_lifecycle COMMAND file_picker_test)
set_tests_properties(asynchronous_file_picker_lifecycle PROPERTIES TIMEOUT 30)
